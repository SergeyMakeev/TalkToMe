cmake_minimum_required(VERSION 3.26)
project(TalkToMeSpeech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_BUILD_TESTS OFF)
set(WHISPER_BUILD_EXAMPLES OFF)
set(WHISPER_BUILD_SERVER OFF)
add_subdirectory("${WHISPER_SOURCE}" whisper EXCLUDE_FROM_ALL)
add_library(TalkToMeSpeech SHARED ../../Shared/WhisperBridge.cpp)
set_target_properties(TalkToMeSpeech PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(TalkToMeSpeech PRIVATE whisper)
