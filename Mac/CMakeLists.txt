cmake_minimum_required(VERSION 3.26)
project(TalkToMe LANGUAGES C CXX Swift)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_Swift_LANGUAGE_VERSION 5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "")
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "")
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "")
set(GGML_BLAS OFF CACHE BOOL "")
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    # Ship a baseline Intel binary including for machines without AVX2.
    foreach(feature AVX AVX2 FMA F16C BMI2)
        set(GGML_${feature} OFF CACHE BOOL "")
    endforeach()
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper EXCLUDE_FROM_ALL)
add_library(whisper_bridge STATIC ../Shared/WhisperBridge.cpp)
target_link_libraries(whisper_bridge PUBLIC whisper)
add_executable(TalkToMe main.swift Preferences.swift AccessibilityAccess.swift Input.swift TextInsertion.swift ClipboardPaste.swift ../Checks/MacInsertionChecks.swift Engine.swift Dictation.swift Views.swift)
target_compile_options(TalkToMe PRIVATE -import-objc-header "${CMAKE_CURRENT_SOURCE_DIR}/../Shared/WhisperBridge.h")
target_link_libraries(TalkToMe PRIVATE whisper_bridge c++ "-framework Carbon")
